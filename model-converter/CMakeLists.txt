cmake_minimum_required(VERSION 3.22)
project(DollViewConvert LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "")
set(USE_STATIC_CRT ON CACHE BOOL "")
set(ASSIMP_BUILD_TESTS OFF CACHE BOOL "")
set(ASSIMP_BUILD_ASSIMP_TOOLS OFF CACHE BOOL "")
set(ASSIMP_BUILD_SAMPLES OFF CACHE BOOL "")
set(ASSIMP_BUILD_DOCS OFF CACHE BOOL "")
set(ASSIMP_INSTALL OFF CACHE BOOL "")
set(ASSIMP_WARNINGS_AS_ERRORS OFF CACHE BOOL "")
set(ASSIMP_BUILD_ZLIB ON CACHE BOOL "")
set(ASSIMP_BUILD_DRACO_STATIC ON CACHE BOOL "")
set(ASSIMP_BUILD_ALL_IMPORTERS_BY_DEFAULT OFF CACHE BOOL "")
set(ASSIMP_BUILD_ALL_EXPORTERS_BY_DEFAULT OFF CACHE BOOL "")
foreach(format FBX GLTF OBJ STL)
  set(ASSIMP_BUILD_${format}_IMPORTER ON CACHE BOOL "")
  set(ASSIMP_BUILD_${format}_EXPORTER ON CACHE BOOL "")
endforeach()
add_subdirectory("${ASSIMP_SOURCE_DIR}" assimp-build EXCLUDE_FROM_ALL)
target_include_directories(assimp PRIVATE "${CMAKE_BINARY_DIR}")
add_executable(DollView.Convert main.cpp textures.cpp)
target_compile_features(DollView.Convert PRIVATE cxx_std_17)
target_include_directories(DollView.Convert PRIVATE "${ASSIMP_SOURCE_DIR}/contrib/stb")
target_link_libraries(DollView.Convert PRIVATE assimp)
